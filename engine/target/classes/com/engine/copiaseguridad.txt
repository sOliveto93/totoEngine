package com.engine;

import java.io.IOException;
import java.io.InputStream;
import java.nio.ByteBuffer;

import org.lwjgl.BufferUtils;
import org.lwjgl.glfw.GLFW;
import org.lwjgl.opengl.GL;
import org.lwjgl.opengl.GL11;
import org.lwjgl.opengl.GL13;
import org.lwjgl.opengl.GL15;
import org.lwjgl.opengl.GL20;
import org.lwjgl.opengl.GL30;

import java.awt.image.BufferedImage;
import java.io.File;
import javax.imageio.ImageIO;

public class Main {

    static String vertexSource;
    static String fragmentSource;
static BufferedImage imagen;
    public static void cargarShaders() {
        try (InputStream vertexInput = Main.class.getResourceAsStream("/shaders/vertex.glsl");
                InputStream fragmentInput = Main.class.getResourceAsStream("/shaders/fragment.glsl");) {

            if (vertexInput == null) {
                throw new IllegalStateException("No se econtro vertex.glsl");
            }
            if (fragmentInput == null) {
                throw new IllegalStateException("No se encontró fragment.glsl");
            }
            vertexSource = new String(vertexInput.readAllBytes());
            fragmentSource = new String(fragmentInput.readAllBytes());
        } catch (IOException e) {
            throw new RuntimeException("Error leyendo los shaders", e);
        }
    }
    public static void cargarRecursos(String path){
        try {
            imagen= ImageIO.read(new File(path));
        } catch (IOException e) {
            throw new RuntimeException("No se pudo cargar la imagen, e");
        }

        System.out.println("Imagen cargada");
System.out.println("Ancho: " + imagen.getWidth());
System.out.println("Alto: " + imagen.getHeight());
    }

    public static void main(String[] args) {

        cargarShaders();
        
       


        System.out.println("Iniciando GLFW...");
        if (!GLFW.glfwInit()) {
            throw new IllegalStateException("No se pudo inicializar GLFW");
        }
        System.out.println("GLFW inicializado");

        // creamos la ventana
        long window = GLFW.glfwCreateWindow(800, 600, "prueba", 0, 0);

        System.out.println("Window handle: " + window);

        if (window == 0) {
            throw new IllegalStateException("No se pudo crear la ventana");
        }

        GLFW.glfwMakeContextCurrent(window);
        GL.createCapabilities();
        

        int vertexShader = GL20.glCreateShader(GL20.GL_VERTEX_SHADER);
        GL20.glShaderSource(vertexShader, vertexSource);
        GL20.glCompileShader(vertexShader);

        int success = GL20.glGetShaderi(vertexShader, GL20.GL_COMPILE_STATUS);

        if (success == GL11.GL_FALSE) {
            String log = GL20.glGetShaderInfoLog(vertexShader);
            throw new IllegalStateException(
                    "Error compilando vertex shader:\n" + log);
        }

        int fragmentShader = GL20.glCreateShader(GL20.GL_FRAGMENT_SHADER);

GL20.glShaderSource(fragmentShader, fragmentSource);

GL20.glCompileShader(fragmentShader);

     success = GL20.glGetShaderi(
        fragmentShader,
        GL20.GL_COMPILE_STATUS
);

if (success == GL11.GL_FALSE) {
    String log = GL20.glGetShaderInfoLog(fragmentShader);
    throw new IllegalStateException(
            "Error compilando fragment shader:\n" + log
    );
}

    int shaderProgram=GL20.glCreateProgram();
GL20.glAttachShader(shaderProgram, vertexShader);
GL20.glAttachShader(shaderProgram, fragmentShader);

GL20.glLinkProgram(shaderProgram);

int positionLocation =
        GL20.glGetUniformLocation(shaderProgram, "positionSprite");

int scaleLocation =
        GL20.glGetUniformLocation(shaderProgram, "scale");

 success = GL20.glGetProgrami(
        shaderProgram,
        GL20.GL_LINK_STATUS
);

if (success == GL11.GL_FALSE) {
    String log = GL20.glGetProgramInfoLog(shaderProgram);

    throw new IllegalStateException(
            "Error linkeando shader program:\n" + log
    );
}





        int vao = GL30.glGenVertexArrays();
        GL30.glBindVertexArray(vao);

        float[] vertices = {
    // posición      // UV
    -0.5f,  0.5f,     0.0f, 0.0f,
     0.5f,  0.5f,     1.0f, 0.0f,
     0.5f, -0.5f,     1.0f, 1.0f,
    -0.5f, -0.5f,     0.0f, 1.0f
};


        int vbo = GL15.glGenBuffers();
        GL15.glBindBuffer(GL15.GL_ARRAY_BUFFER, vbo);

        GL15.glBufferData(GL15.GL_ARRAY_BUFFER, vertices, GL15.GL_STATIC_DRAW);


/*EBO para retulizar y decir como se unen los vertices*/
int[] indices={
    0,1,2,
    2,3,0
};


int ebo=GL15.glGenBuffers();

GL15.glBindBuffer(GL15.GL_ELEMENT_ARRAY_BUFFER, ebo);

GL15.glBufferData(GL15.GL_ELEMENT_ARRAY_BUFFER,
    indices,
    GL15.GL_STATIC_DRAW);
/******** */



//location 0 -> x,y
        GL20.glVertexAttribPointer(0, 2, GL11.GL_FLOAT, false, 4 * Float.BYTES, 0);
        
        GL20.glEnableVertexAttribArray(0);
//location 1 -> u,v
        GL20.glVertexAttribPointer(1,2,GL11.GL_FLOAT, false, 4*Float.BYTES, 2*Float.BYTES);
GL20.glEnableVertexAttribArray(1);


        GLFW.glfwShowWindow(window);

        System.out.println("Ventana mostrada");
        System.out.println(
                "Visible: " +
                        GLFW.glfwGetWindowAttrib(window, GLFW.GLFW_VISIBLE));

cargarRecursos("engine/src/main/resources/textures/tileset.png");
 
int width= imagen.getWidth();
int height=imagen.getHeight();

ByteBuffer buffer = BufferUtils.createByteBuffer(width*height*4);

for (int y = 0; y < height; y++) {
    for (int x = 0; x < width; x++) {

        int pixel = imagen.getRGB(x, y);

        byte red   = (byte) ((pixel >> 16) & 0xFF);
        byte green = (byte) ((pixel >> 8) & 0xFF);
        byte blue  = (byte) (pixel & 0xFF);
        byte alpha = (byte) ((pixel >> 24) & 0xFF);

        buffer.put(red);
        buffer.put(green);
        buffer.put(blue);
        buffer.put(alpha);
    }
}
buffer.flip();

 int texture= GL11.glGenTextures();
 GL11.glBindTexture(GL11.GL_TEXTURE_2D, texture);

int uvMinLocation =
    GL20.glGetUniformLocation(shaderProgram, "uvMin");

int uvMaxLocation =
    GL20.glGetUniformLocation(shaderProgram, "uvMax");


GL11.glTexParameteri(
    GL11.GL_TEXTURE_2D,
    GL11.GL_TEXTURE_MIN_FILTER,
    GL11.GL_NEAREST
);

GL11.glTexParameteri(
    GL11.GL_TEXTURE_2D,
    GL11.GL_TEXTURE_MAG_FILTER,
    GL11.GL_NEAREST
);

GL11.glTexImage2D(GL11.GL_TEXTURE_2D,0,GL11.GL_RGBA,width,height,0,GL11.GL_RGBA,GL11.GL_UNSIGNED_BYTE,
    buffer);

int samplerLocation =
    GL20.glGetUniformLocation(
        shaderProgram,
        "textureSampler"
    );

        while (!GLFW.glfwWindowShouldClose(window)) {


            GL11.glClearColor(0.2f, 0.4f, 0.8f, 1.0f);
            GL11.glClear(  GL11.GL_COLOR_BUFFER_BIT );
            
          
            
            GL20.glUseProgram(shaderProgram);

GL20.glUniform2f(uvMinLocation, 0.25f, 0.0f);
GL20.glUniform2f(uvMaxLocation, 0.375f, 0.125f);


GL20.glUniform2f(positionLocation, 0.10f, 0.60f);
GL20.glUniform2f(scaleLocation, 0.08f, 0.10666f);

            GL13.glActiveTexture(GL13.GL_TEXTURE0);
           
GL11.glBindTexture(
    GL11.GL_TEXTURE_2D,
    texture
);
GL20.glUniform1i(
    samplerLocation,
    0
);

GL11.glDrawElements(
         GL11.GL_TRIANGLES,
    6,
    GL11.GL_UNSIGNED_INT,
    0
);
            

GL20.glUniform2f(uvMinLocation, 0.25f, 0.0f);
GL20.glUniform2f(uvMaxLocation, 0.375f, 0.125f);

GL20.glUniform2f(positionLocation, 0.5f, 0.0f);

GL11.glDrawElements(
    GL11.GL_TRIANGLES,
    6,
    GL11.GL_UNSIGNED_INT,
    0
);
            
            GLFW.glfwSwapBuffers(window);
            GLFW.glfwPollEvents();
        }

        System.out.println("Cerrando...");





        GLFW.glfwDestroyWindow(window);
        GLFW.glfwTerminate();
        GL30.glDeleteVertexArrays(vao);
GL15.glDeleteBuffers(vbo);
GL15.glDeleteBuffers(ebo);
GL20.glDeleteShader(vertexShader);
GL20.glDeleteShader(fragmentShader);
GL20.glDeleteProgram(shaderProgram);

GLFW.glfwDestroyWindow(window);
GLFW.glfwTerminate();

    }
}
